\subsection*{Bases, dimensions, and complements}

% MA 3.23(a-d), original data.
\begin{problem}[Selecting columns]\label{ex:03:chosen-columns}
Let \(a_1,\ldots,a_9\) be the columns of
\[
A=\begin{pmatrix}1&1&1&1&1&1&2&2&5\\0&1&1&2&2&3&-1&3&3\\1&1&2&3&5&5&1&0&4\end{pmatrix}.
\]
Which lists are bases of \(\R^3\)?
\begin{enumerate}
\item \((a_1,a_2)\).
\item \((a_1,a_4,a_6,a_8)\).
\item \((a_2,a_4,a_7)\).
\item \((a_3,a_5,a_9)\).
\end{enumerate}
\end{problem}
\begin{solution}
(a) and (b) have the wrong lengths, since every basis of \(\R^3\) has three members.
For (c), a zero combination gives
\[
s+t+2u=0,\qquad s+2t-u=0,\qquad s+3t+u=0.
\]
Successive differences give \(t-3u=0\) and \(t+2u=0\); hence
\(u=t=s=0\). Three independent vectors form a basis of \(\R^3\).
For (d), the nonzero coefficient relation
\(7a_3-2a_5-a_9=0\) proves dependence.
\end{solution}

% Bapat 1.14(i-iii); no supplied answer.
\begin{problem}[Combining a given basis]\label{ex:03:combining-basis}
If \((x,y,z)\) is a basis of \(\R^3\), which lists are also bases?
\begin{enumerate}
\item \((x+2y,\ y+3z,\ x+2z)\).
\item \((x+y-2z,\ x-2y+z,\ -2x+y+z)\).
\item \((x,\ y,\ x+y+z)\).
\end{enumerate}
\end{problem}
\begin{solution}
For (a), a zero combination with coefficients \(a,b,c\) gives
\[
(a+c)x+(2a+b)y+(3b+2c)z=0.
\]
Thus \(c=-a\), \(b=-2a\), and \(-8a=0\); all coefficients vanish.
The list is a basis. In (b) the three displayed vectors sum to zero,
so they are dependent and not a basis. For (c),
\[
ax+by+c(x+y+z)=(a+c)x+(b+c)y+cz=0
\]
forces \(c=a=b=0\); the list is a basis.
\end{solution}

% Bapat 1.15(i,ii); coefficient proof supplied.
\begin{problem}[Mixing two bases]\label{ex:03:mixed-bases}
Suppose \((x_1,x_2)\) and \((y_1,y_2)\) are bases of \(\R^2\).
Prove that at least one of the following statements holds:
\begin{enumerate}
\item Both \((x_1,y_2)\) and \((x_2,y_1)\) are bases.
\item Both \((x_1,y_1)\) and \((x_2,y_2)\) are bases.
\end{enumerate}
\end{problem}
\begin{solution}
Write \(y_1=ax_1+bx_2\), \(y_2=cx_1+dx_2\).
Comparison of coefficients in the \(x\)-basis shows
\[
(x_1,y_2)\text{ is independent}\iff d\ne0,\qquad
(x_2,y_1)\text{ is independent}\iff a\ne0.
\]
The analogous conditions for the two pairs in (b) are \(b\ne0\) and \(c\ne0\).
If \(a,d\ne0\), (a) holds. If \(a=0\), then \(b\ne0\) since \(y_1\ne0\);
also \(c\ne0\), since otherwise both \(y_1,y_2\) would lie in \(\Span(x_2)\).
If \(d=0\), then \(c\ne0\) and the same argument forces \(b\ne0\).
Thus whenever (a) fails, (b) holds.
\end{solution}

% Bapat 1.13(i,ii).
\begin{problem}[Spaces given by equations]\label{ex:03:constraint-dimensions}
Verify that each set is a subspace and find a basis and its dimension:
\begin{enumerate}
\item \(\{(a,b,c,d)^\top:a+b=c+d\}\subseteq\R^4\).
\item \(\{(x,y,z)^\top:2x-y=0, 2y+3z=0\}\subseteq\R^3\).
\end{enumerate}
\end{problem}
\begin{solution}
Both are solution sets of homogeneous linear systems, hence subspaces
by Unit 1, Proposition~\ref{one-prop:01:closure}.
In (a), solve \(a=-b+c+d\), obtaining exactly
\[
b\begin{pmatrix}-1\\1\\0\\0\end{pmatrix}
+c\begin{pmatrix}1\\0\\1\\0\end{pmatrix}
+d\begin{pmatrix}1\\0\\0\\1\end{pmatrix}.
\]
Coordinates 2, 3, and 4 establish independence, so the dimension is 3.
In (b), \(y=2x\) and \(z=-4x/3\); the set is
\(\{t(3,6,-4)^\top:t\in\R\}\). The displayed nonzero column is a basis,
and the dimension is 1.
\end{solution}

% Bapat 1.16.
\begin{problem}[Zero-sum columns]\label{ex:03:zero-sum}
For \(n\geqslant1\), find a basis of
\(W=\{x\in\R^n:\sum_{i=1}^n x_i=0\}\) and verify that it is a subspace.
\end{problem}
\begin{solution}
Zero belongs to \(W\), and the coordinate sum of \(sx+ty\) is
\(s\sum_i x_i+t\sum_i y_i=0\) for \(x,y\in W\).
Solving for the last coordinate gives
\[
x=\sum_{i=1}^{n-1}x_i(e_i-e_n).
\]
The first \(n-1\) coordinates of a zero combination force all coefficients
to vanish. Thus \((e_1-e_n,\ldots,e_{n-1}-e_n)\) is a basis and
\(\dim W=n-1\). For \(n=1\), this is the empty basis of \(\{0\}\).
\end{solution}

% Bapat 1.17; Chapter 12 gives only n^2-1.
\begin{problem}[Trace-zero matrices]\label{ex:03:trace-zero}
Determine the dimension of \(W=\{A\in\R^{n\times n}:\tr A=0\}\),
where \(n\geqslant1\), by exhibiting a basis.
\end{problem}
\begin{solution}
Linearity of trace proves that \(W\) is a subspace. A basis is
\[
\{E_{ij}:i\ne j\}\ \cup\ \{E_{ii}-E_{nn}:1\leqslant i<n\}.
\]
Indeed, \(a_{nn}=-\sum_{i<n}a_{ii}\), so every \(A\in W\) equals
\[
\sum_{i\ne j}a_{ij}E_{ij}+\sum_{i<n}a_{ii}(E_{ii}-E_{nn}).
\]
Reading each off-diagonal entry and then diagonal entries \(1,\ldots,n-1\)
proves independence. The dimension is \(n(n-1)+(n-1)=n^2-1\).
For \(n=1\), \(W=\{0\}\), consistently with the empty basis.
\end{solution}

% MA 3.24(a,b), original five-parameter representation retained.
\begin{problem}[Equal row and column sums]\label{ex:03:row-column-sums}
Find bases for the spaces of (a) \(2\times2\) and (b) \(3\times3\)
real matrices whose row sums and column sums are all equal.
The common sum may depend on the matrix.
\end{problem}
\begin{solution}
The defining equalities are homogeneous linear equations in the entries,
so both sets are subspaces. For (a), comparison of the two row sums
with the column sums gives precisely
\[
\begin{pmatrix}a&b\\b&a\end{pmatrix}
=a\begin{pmatrix}1&0\\0&1\end{pmatrix}
+b\begin{pmatrix}0&1\\1&0\end{pmatrix}.
\]
The two coefficient matrices form a basis: their first rows distinguish
the coefficients. The dimension is 2.

For (b), write the top left \(2\times2\) block as
\(\left(\begin{smallmatrix}a&b\\c&d\end{smallmatrix}\right)\), and let
the bottom right entry be \(-e\). If the common sum is \(s\), the
remaining entries in the first two rows and columns are
\[
a_{13}=s-a-b,\quad a_{23}=s-c-d,\quad
a_{31}=s-a-c,\quad a_{32}=s-b-d.
\]
The last column sum gives \(s=a+b+c+d+e\). Thus the complete family is
\[
\begin{pmatrix}
a&b&c+d+e\\c&d&a+b+e\\b+d+e&a+c+e&-e
\end{pmatrix}.
\]
Conversely, each of its six sums is \(a+b+c+d+e\).
The five coefficient matrices are
\begin{gather*}
M_a=\begin{pmatrix}1&0&0\\0&0&1\\0&1&0\end{pmatrix},\quad
M_b=\begin{pmatrix}0&1&0\\0&0&1\\1&0&0\end{pmatrix},\quad
M_c=\begin{pmatrix}0&0&1\\1&0&0\\0&1&0\end{pmatrix},\\
M_d=\begin{pmatrix}0&0&1\\0&1&0\\1&0&0\end{pmatrix},\qquad
M_e=\begin{pmatrix}0&0&1\\0&0&1\\1&1&-1\end{pmatrix}.
\end{gather*}
They span the family. In a zero combination, entries \((1,1),(1,2),(2,1),(2,2),(3,3)\)
give the five coefficients as zero. They are a basis; the dimension is 5.
\end{solution}

% Harville Exercises and Solutions 4.8(a-d).
\begin{problem}[Extracting a basis with prescribed members]\label{ex:03:prescribed-basis}
Let
\[
A=\begin{pmatrix}0&1&0&-3&2\\0&-2&0&6&2\\0&2&2&5&2\\0&-4&-2&1&0\end{pmatrix},\qquad
u=\begin{pmatrix}2\\-1\\3\\-4\end{pmatrix},\quad
v=\begin{pmatrix}0\\9\\-3\\12\end{pmatrix}.
\]
\begin{enumerate}
\item Express \(u,v\) as combinations of the columns of \(A\).
\item Extract a basis of \(\mathcal C(A)\) by scanning the columns in order.
\item Find \(\rank A\).
\item Find a basis of \(\mathcal C(A)\) that includes \(u,v\).
\end{enumerate}
\end{problem}
\begin{solution}
Write \(a_j\) for column \(j\).
(a) Direct comparison gives \(u=a_2+\tfrac12a_5\),
\(v=-3a_2+\tfrac32a_5\).

(b) Discard \(a_1=0\), retain \(a_2,a_3\), and discard
\(a_4=-3a_2+\tfrac{11}{2}a_3\).
The column \(a_5\) must be retained. In fact, if
\(\alpha a_2+\beta a_3+\gamma a_5=0\), the first two coordinates give
\(\alpha+2\gamma=0\), \(-2\alpha+2\gamma=0\), so
\(\gamma=\alpha=0\); the third gives \(2\beta=0\).
Thus \((a_2,a_3,a_5)\) is the extracted basis.

(c) This basis has three members, hence \(\rank A=3\).

(d) In a relation \(su+tv+wa_3=0\), substitute (a):
\[
(s-3t)a_2+w a_3+(s/2+3t/2)a_5=0.
\]
Part (b) gives \(s=3t\), \(w=0\), \(s+3t=0\); hence \(s=t=w=0\).
The three independent members of the three-dimensional space form a basis.
For an explicit spanning check,
\(a_2=u/2-v/6\) and \(a_5=u+v/3\), so every original column lies in
\(\Span(u,v,a_3)\).
\end{solution}

% Bapat 2.2 and 2.3, each complete.
\begin{problem}[Basis exchange]\label{ex:03:exchange}
\begin{enumerate}
\item If \((x_1,\ldots,x_p)\) and \((y_1,\ldots,y_q)\) are independent
lists in \(\R^n\), with \(p<q\), show that some \(y_i\) can be adjoined
to the first list while preserving independence.
\item Let \(X,Y\) be bases of \(\R^n\). If \(S\subseteq X\) contains
\(r\) members, show that there is a subset \(T\subseteq Y\) of \(r\)
members for which \((X\setminus S)\cup T\) is a basis.
\end{enumerate}
\end{problem}
\begin{solution}
(a) If no \(y_i\) can be adjoined, Proposition~\ref{prop:03:adjoin} places
every \(y_i\) in \(\Span(x_1,\ldots,x_p)\).
This gives \(q>p\) independent vectors in a space spanned by \(p\) vectors,
contradicting Theorem~\ref{thm:03:bound}.

(b) Start with the \(n-r\) independent members of \(X\setminus S\).
While the current list has fewer than \(n\) members, apply (a) with
the \(n\)-member basis \(Y\), adjoining one of its members outside
the current span. No member can be chosen twice.
After \(r\) additions the list is independent of length \(n\), hence a basis.
The added members form the required \(T\). For \(r=0\), use \(T=\varnothing\).
\end{solution}

\begin{problem}[A product space]\label{ex:03:product-space}
For subspaces \(U\subseteq\R^m\) and \(W\subseteq\R^n\), define
\[
U\times W=\left\{\begin{pmatrix}u\\w\end{pmatrix}:u\in U, w\in W\right\}.
\]
Given bases \((u_1,\ldots,u_r)\), \((w_1,\ldots,w_s)\), find a basis of
\(U\times W\) and prove \(\dim(U\times W)=r+s\).
\end{problem}
\begin{solution}
The set is a subspace of \(\R^{m+n}\) by componentwise closure.
A basis is
\[
\begin{pmatrix}u_1\\0\end{pmatrix},\ldots,\begin{pmatrix}u_r\\0\end{pmatrix},
\begin{pmatrix}0\\w_1\end{pmatrix},\ldots,\begin{pmatrix}0\\w_s\end{pmatrix}.
\]
It spans because the two components can be expanded separately.
In a zero combination, the upper block gives \(\sum_i a_iu_i=0\),
and the lower block gives \(\sum_j b_jw_j=0\). Both original lists
are independent, so all coefficients vanish. The basis has \(r+s\) members.
\end{solution}

% Bapat 2.1; Chapter 12 answer verified by parameter-safe elimination.
\begin{problem}[An exceptional parameter]\label{ex:03:parameter-rank}
Find the rank, for every \(\alpha\in\R\), of
\[
A_\alpha=\begin{pmatrix}1&4&\alpha&4\\2&-6&7&1\\3&2&-6&7\\2&2&-5&5\end{pmatrix}.
\]
\end{problem}
\begin{solution}
Subtract \(2R_1,3R_1,2R_1\) from rows 2, 3, 4, respectively:
\[
\begin{pmatrix}
1&4&\alpha&4\\0&-14&7-2\alpha&-7\\
0&-10&-6-3\alpha&-5\\0&-6&-5-2\alpha&-3
\end{pmatrix}.
\]
Next replace \(R_3\) by \(7R_3-5R_2\), \(R_4\) by \(7R_4-3R_2\),
and then \(R_4\) by \(11R_4-8R_3\), obtaining
\[
\begin{pmatrix}
1&4&\alpha&4\\0&-14&7-2\alpha&-7\\
0&0&-11(\alpha+7)&0\\0&0&0&0
\end{pmatrix}.
\]
Each replacement is a nonzero constant scaling followed by a row addition,
so it is valid for every \(\alpha\). The first two pivots never vanish.
Therefore \(\rank A_\alpha=2\) for \(\alpha=-7\), and 3 otherwise.
\end{solution}

% Bapat 2.6, no supplied answer.
\begin{problem}[Two rank factorizations]\label{ex:03:two-factorizations}
Find two different rank factorizations of
\[
A=\begin{pmatrix}1&1&2&0\\2&-3&1&1\\3&-2&3&1\\5&-5&4&2\end{pmatrix}.
\]
\end{problem}
\begin{solution}
The first two columns \(a_1,a_2\) are independent: a zero combination
gives \(s+t=0\), \(2s-3t=0\), hence \(s=t=0\).
The remaining columns satisfy
\(a_3=(7a_1+3a_2)/5\), \(a_4=(a_1-a_2)/5\). Thus
\[
A=\underbrace{\begin{pmatrix}1&1\\2&-3\\3&-2\\5&-5\end{pmatrix}}_B
\underbrace{\begin{pmatrix}1&0&7/5&1/5\\0&1&3/5&-1/5\end{pmatrix}}_F.
\]
The factors have rank 2, so this is a rank factorization.
Choose \(a_1,a_4\) instead. Since \(a_2=a_1-5a_4\) and
\(a_3=2a_1-3a_4\),
\[
A=\begin{pmatrix}1&0\\2&1\\3&1\\5&2\end{pmatrix}
\begin{pmatrix}1&1&2&0\\0&-5&-3&1\end{pmatrix}.
\]
The first factor has independent columns by its first two rows;
columns 1 and 4 of the second factor form \(I_2\), proving its rank is 2.
\end{solution}

% MA 4.6(a), second matrix only, and (c); first matrix is Example 3.5.
\begin{problem}[A rectangular rank calculation]\label{ex:03:rectangular-rank}
Find the rank of
\[
B=\begin{pmatrix}1&5&6&3\\2&1&4&1\\3&5&5&4\end{pmatrix}.
\]
Can any \(3\times4\) matrix have rank 4?
\end{problem}
\begin{solution}
Use \(R_2\leftarrow R_2-2R_1\), \(R_3\leftarrow R_3-3R_1\),
and then \(R_3\leftarrow9R_3-10R_2\):
\[
B\longrightarrow
\begin{pmatrix}1&5&6&3\\0&-9&-8&-5\\0&-10&-13&-5\end{pmatrix}
\longrightarrow
\begin{pmatrix}1&5&6&3\\0&-9&-8&-5\\0&0&-37&5\end{pmatrix}.
\]
There are three pivots, so \(\rank B=3\).
Every \(3\times4\) matrix has its column space inside \(\R^3\),
so its rank cannot exceed 3.
\end{solution}

% MA 4.9(a,b), both original counterexamples.
\begin{problem}[Diagonal entries and rank]\label{ex:03:diagonal-rank}
Prove that a diagonal matrix has rank equal to the number of its nonzero
diagonal entries. Does the same statement hold for triangular matrices?
\end{problem}
\begin{solution}
The nonzero columns of a diagonal matrix are nonzero multiples of distinct
standard columns, hence independent; all other columns are zero.
For triangular matrices, consider
\[
L=\begin{pmatrix}0&0\\1&0\end{pmatrix},\qquad
M=\begin{pmatrix}1&0&0\\0&0&0\\0&1&0\end{pmatrix}.
\]
The first has rank 1 and no nonzero diagonal entry. The second has rank 2,
from its first two independent columns, but only one nonzero diagonal entry.
\end{solution}

% MA 4.10, both directions; explicit nonzero condition retained.
\begin{problem}[Rank one]\label{ex:03:rank-one}
Show that a real \(m\times n\) matrix \(A\) has rank 1 if and only if
\(A=ab^\top\) for some nonzero \(a\in\R^m\), \(b\in\R^n\).
\end{problem}
\begin{solution}
If \(A=ab^\top\), its columns are \(b_ja\). They lie in \(\Span(a)\),
and some \(b_j\ne0\), so one column spans that one-dimensional space.
Conversely, if \(\rank A=1\), choose a nonzero column \(a\).
It is a basis of \(\mathcal C(A)\), so each column is \(b_ja\).
Collecting these coefficients gives \(A=ab^\top\); the coefficient of
the chosen column is 1, ensuring \(b\ne0\).
\end{solution}

% MA 4.11(b), Harville 4.10; minimality is an explicitly added consequence.
\begin{problem}[Rank as a number of outer products]\label{ex:03:outer-sum}
If \(\rank A=r\), express \(A\) as the sum of \(r\) rank-one matrices.
Also show that fewer than \(r\) such summands cannot suffice.
\end{problem}
\begin{solution}
For \(r>0\), take a rank factorization \(A=BF\), write the columns of
\(B\) as \(b_i\) and the rows of \(F\) as \(f_i^\top\), and expand:
\[
A=\sum_{i=1}^r b_if_i^\top.
\]
The columns of \(B\) and rows of \(F\) are independent, so none is zero.
Exercise~\ref{ex:03:rank-one} makes each summand rank one.
If instead \(A=\sum_{i=1}^k u_iv_i^\top\), every column of \(A\)
is a combination of \(u_1,\ldots,u_k\), whence \(\rank A\leqslant k\).
Thus \(k\geqslant r\). For \(A=0\), the empty sum has zero summands.
\end{solution}

% MA 4.4*, full original lifting proof; alternative to the echelon proof in the text.
\begin{problem}[\(\ast\) Lifting an image basis]\label{ex:03:lifted-basis}
Let \(x_1,\ldots,x_r\) be a basis of \(\mathcal C(A)\), and
\(y_1,\ldots,y_s\) a basis of \(N(A)\), where \(A:m\times n\).
For each \(j\), choose \(z_j\) with \(Az_j=x_j\).
Prove directly that \((y_1,\ldots,y_s,z_1,\ldots,z_r)\) is a basis of
\(\R^n\), and deduce \(r+s=n\).
\end{problem}
\begin{solution}
\begin{itemize}
\item For \(v\in\R^n\), expand \(Av=\sum_j a_jx_j\).
Then
\[
A\left(v-\sum_j a_jz_j\right)=Av-\sum_j a_jx_j=0,
\]
so \(v-\sum_j a_jz_j=\sum_i b_iy_i\) for suitable \(b_i\).
Thus \(v=\sum_i b_iy_i+\sum_j a_jz_j\), proving spanning.
\item In a zero combination
\(\sum_i b_iy_i+\sum_j a_jz_j=0\), apply \(A\) to obtain
\(\sum_j a_jx_j=0\). Independence of the \(x_j\) gives all \(a_j=0\);
independence of the \(y_i\) then gives all \(b_i=0\).
\item The combined list is a basis, so its length is \(n\).
The argument includes empty lists when \(r=0\) or \(s=0\).
\end{itemize}
\end{solution}

% MA 4.12.
\begin{problem}[An additional column]\label{ex:03:additional-column}
Suppose \(A:m\times n\) and there is \(a\in\R^m\) such that
\(\rank[\,A\ a\,]=\rank A+1\). Prove that the rows of \(A\) are dependent.
\end{problem}
\begin{solution}
There are only \(m\) rows in the augmented matrix, so
\(\rank A+1\leqslant m\). Thus \(\rank A<m\), and its \(m\) rows
cannot be independent.
\end{solution}

% MA 4.14(a,b); source containment wording corrected.
\begin{problem}[Ranks of sums and differences]\label{ex:03:rank-sum}
For matrices \(A,B\) of the same size, prove
\[
\rank(A+B)\leqslant\rank A+\rank B,\qquad
\rank(A-B)\geqslant|\rank A-\rank B|.
\]
\end{problem}
\begin{solution}
Every column of \(A+B\) belongs to \(\mathcal C(A)+\mathcal C(B)\).
By the dimension formula,
\[
\rank(A+B)\leqslant\dim\bigl(\mathcal C(A)+\mathcal C(B)\bigr)
\leqslant\rank A+\rank B.
\]
Apply this result to \(A=(A-B)+B\) to obtain
\(\rank(A-B)\geqslant\rank A-\rank B\).
Interchange \(A,B\), noting that multiplication by \(-1\) leaves a span
unchanged, to get the other lower bound.
\end{solution}

% MA 4.17; use row/column inclusions to avoid U4 product-rank theorem.
\begin{problem}[Deleting rows and columns]\label{ex:03:submatrix}
If \(A_1\) is a submatrix of \(A\), prove \(\rank A_1\leqslant\rank A\).
\end{problem}
\begin{solution}
Delete the unwanted rows first, obtaining \(B\). Its row space is spanned
by a subset of the rows of \(A\), so \(\rank B\leqslant\rank A\).
Now delete the unwanted columns of \(B\). The remaining columns span
a subspace of \(\mathcal C(B)\), so \(\rank A_1\leqslant\rank B\).
Combining the inequalities proves the result.
\end{solution}

\subsection*{Block structure and rank}

% MA 5.41(a,b); independence checked for the full embedded bases.
\begin{problem}[Two zero blocks]\label{ex:03:block-diagonal}
For arbitrary rectangular blocks, with zero blocks of the required sizes, prove
\[
\rank\begin{pmatrix}A&0\\0&D\end{pmatrix}=\rank A+\rank D,
\qquad
\rank\begin{pmatrix}0&B\\C&0\end{pmatrix}=\rank B+\rank C.
\]
\end{problem}
\begin{solution}
Take bases \(a_1,\ldots,a_r\) of \(\mathcal C(A)\) and
\(d_1,\ldots,d_s\) of \(\mathcal C(D)\). The columns
\[
\begin{pmatrix}a_i\\0\end{pmatrix}\ (1\leqslant i\leqslant r),\qquad
\begin{pmatrix}0\\d_j\end{pmatrix}\ (1\leqslant j\leqslant s)
\]
span the block matrix's column space. A zero combination has upper block
\(\sum_i c_ia_i=0\) and lower block \(\sum_j e_jd_j=0\), so all
coefficients vanish. Its rank is \(r+s\).
For the second identity, interchange the two column groups to obtain
\(\left(\begin{smallmatrix}B&0\\0&C\end{smallmatrix}\right)\).
A reordering of columns does not change their span, so the first identity applies.
\end{solution}

% MA 5.42, both patterns.
\begin{problem}[A single zero block is insufficient]\label{ex:03:block-counterexample}
Show that neither of the identities
\[
\rank\begin{pmatrix}A&B\\0&D\end{pmatrix}=\rank A+\rank D,
\qquad
\rank\begin{pmatrix}A&0\\C&D\end{pmatrix}=\rank A+\rank D
\]
holds in general.
\end{problem}
\begin{solution}
Take \(A=D=[0]\) and \(B=C=[1]\).
Each block matrix has one nonzero column and rank 1, whereas
\(\rank A+\rank D=0\). More generally, if the diagonal blocks vanish,
the respective ranks are \(\rank B\) and \(\rank C\).
\end{solution}

% MA 5.53; direct row basis proof replaces later block-rank formulas.
\begin{problem}[A bordered matrix]\label{ex:03:border}
For \(A\in\R^{m\times n}\), \(a\in\R^n\), and \(\alpha\in\R\), let
\[
Z=\begin{pmatrix}0&A\\\alpha&a^\top\end{pmatrix}.
\]
Prove
\[
\rank Z=\begin{cases}
\rank A,&\alpha=0\text{ and }a\in\mathcal R(A),\\
\rank A+1,&\text{otherwise}.
\end{cases}
\]
\end{problem}
\begin{solution}
The transposed first \(m\) rows span
\[
U=\left\{\begin{pmatrix}0\\r\end{pmatrix}:r\in\mathcal R(A)\right\}.
\]
If \((r_1,\ldots,r_k)\) is a basis of \(\mathcal R(A)\), the columns
\(\binom{0}{r_i}\) form a basis of \(U\): their lower blocks prove
both spanning and independence. Thus \(\dim U=\rank A\).
The transposed last row is \(z=\binom{\alpha}{a}\), and
\[
z\in U\quad\Longleftrightarrow\quad \alpha=0\text{ and }a\in\mathcal R(A).
\]
Adjoining \(z\) leaves the dimension unchanged in this case and increases
it by one otherwise, by Proposition~\ref{prop:03:adjoin}.
\end{solution}

\begin{problem}[Equivalence of rectangular matrices]\label{ex:03:equivalence}
For \(A,B\in\R^{m\times n}\), prove that
\[
B=PAQ\text{ for some invertible }P,Q
\quad\Longleftrightarrow\quad\rank A=\rank B.
\]
\end{problem}
\begin{solution}
\begin{itemize}
\item The forward direction is Proposition~\ref{prop:03:rank-invariance}.
\item For the converse, first prove that every rank-\(r\) matrix \(A\) can be
changed to \(J_r=\left(\begin{smallmatrix}I_r&0\\0&0\end{smallmatrix}\right)\)
by invertible multiplication.
For \(r>0\), take \(A=CF\) with full rank factors. Extend the columns
of \(C\) to a basis of \(\R^m\), forming an invertible matrix \(M\).
Extend the transposes of the rows of \(F\) to a basis of \(\R^n\);
use their transposes as the rows of an invertible matrix \(N\).
Invertibility follows from the square criterion in Unit 2 and equality
of row and column ranks. Then
\[
M^{-1}C=\begin{pmatrix}I_r\\0\end{pmatrix},\qquad
FN^{-1}=[\,I_r\ 0\,],\qquad M^{-1}AN^{-1}=J_r.
\]
For \(r=0\), \(A=0=J_0\), so identity multipliers suffice.

\item Apply this construction to \(A,B\) of the same rank:
\(P_1AQ_1=J_r=P_2BQ_2\). Therefore
\[
B=P_2^{-1}P_1A Q_1Q_2^{-1}.
\]
Both new multipliers are invertible by the product inverse law.
\end{itemize}
\end{solution}

\begin{problem}[Stacking rows]\label{ex:03:stacking}
For \(A:m\times n\), \(C:q\times n\), and \(D=\left(\begin{smallmatrix}A\\C\end{smallmatrix}\right)\), prove
\begin{enumerate}
\item \(\mathcal R(D)=\mathcal R(C)\) if and only if \(\mathcal R(A)\subseteq\mathcal R(C)\).
\item \(\rank D\geqslant\rank C\), with equality exactly under the same inclusion.
\end{enumerate}
\end{problem}
\begin{solution}
The rows of \(D\) consist of the two lists of rows, so
\(\mathcal R(D)=\mathcal R(A)+\mathcal R(C)\).
This equals \(\mathcal R(C)\) exactly when \(\mathcal R(A)\subseteq\mathcal R(C)\),
proving (a). The inclusion \(\mathcal R(C)\subseteq\mathcal R(D)\)
gives (b)'s inequality. For nested finite-dimensional subspaces, equal
dimensions are equivalent to equal spaces, proving the equality criterion.
\end{solution}

\begin{problem}[Changing a few entries]\label{ex:03:changed-entries}
If \(B\) is obtained from \(A\in\R^{m\times n}\) by changing \(k\)
entries, prove
\[
\rank A-k\leqslant\rank B\leqslant\rank A+k.
\]
\end{problem}
\begin{solution}
Let \(D=B-A\). At most \(k\) entries of \(D\) are nonzero, so at most
\(k\) rows are nonzero; its row space is spanned by at most \(k\) vectors.
Thus \(\rank D\leqslant k\).
By Exercise~\ref{ex:03:rank-sum},
\[
\rank B=\rank(A+D)\leqslant\rank A+k,
\qquad
\rank A=\rank(B-D)\leqslant\rank B+k.
\]
Rearranging the second inequality gives the lower bound.
\end{solution}
