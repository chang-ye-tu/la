\subsection*{Dot products and norms}

% MA 1.8, complete.
\begin{problem}[Computing dot products]\label{ex:06:dot-products}
Let
\[
x=(1,2,-3)^\top,\quad y=(4,-5,1)^\top,\quad
z=(1,1,1)^\top,\quad w=(3,\alpha,1)^\top.
\]
Compute \(x^\top y,x^\top z,y^\top z\), and find all \(\alpha\)
such that \(y^\top w=0\).
\end{problem}
\begin{solution}
Directly,
\[
x^\top y=4-10-3=-9,\qquad x^\top z=1+2-3=0,\qquad
y^\top z=4-5+1=0.
\]
Since \(y^\top w=12-5\alpha+1=13-5\alpha\), the last condition
holds exactly when \(\alpha=13/5\).
\end{solution}

% MA 1.11(a-d), complete.
\begin{problem}[Normalization]\label{ex:06:normalization}
For \(x\ne0\), put \(x_\circ=x/\|x\|\).
\begin{enumerate}
\item Prove that \(\|x_\circ\|=1\).
\item Find the norms of \(a=(1,2)^\top,b=(1,0)^\top,c=(3,4)^\top\).
\item Normalize \(a,b,c\).
\item Prove that \(x\) and \(\lambda x\) have the same normalized vector
for every \(\lambda>0\).
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item Positivity gives \(\|x\|>0\), and homogeneity gives
\(\|x/\|x\|\|=\|x\|/\|x\|=1\).
\item The squared norms are \(1+4=5,1,9+16=25\), respectively.
Thus the norms are \(\sqrt5,1,5\).
\item The answers are
\[
a_\circ=\frac1{\sqrt5}(1,2)^\top,\qquad
b_\circ=(1,0)^\top,\qquad c_\circ=\frac15(3,4)^\top.
\]
\item Since \(\|\lambda x\|=|\lambda|\|x\|\),
\[
(\lambda x)_\circ=\frac{\lambda}{|\lambda|}\,x_\circ=x_\circ
\qquad(\lambda>0).
\]
For \(\lambda<0\) this formula gives \(-x_\circ\);
for \(\lambda=0\) normalization is undefined.
\end{enumerate}
\end{solution}

% MA 1.12(a,b); (c) is proved in the main text.
\begin{problem}[All orthogonal columns]\label{ex:06:orthogonal-r2}
Find all columns of \(\R^2\) orthogonal to
(a) \(a=(1,2)^\top\), and (b) \(b=(1,0)^\top\).
\end{problem}
\begin{solution}
Writing \(x=(x_1,x_2)^\top\), condition (a) is \(x_1+2x_2=0\).
All its solutions are \(x=\lambda(2,-1)^\top\), \(\lambda\in\R\).
Condition (b) is \(x_1=0\), whose solutions are
\(x=\lambda(0,1)^\top\).
Substitution verifies each displayed family; the scalar equations show
that no other columns occur.
\end{solution}

% MA 1.14, complete; the second original counterexample uses MA 1.8.
\begin{problem}[Failure of transitivity]\label{ex:06:nontransitivity}
Does \(x\perp y\) and \(y\perp z\) imply \(x\perp z\)?
Give a counterexample if the implication fails.
\end{problem}
\begin{solution}
Take the three columns, in this order,
\[
x=(1,2,-3)^\top,\qquad y=(1,1,1)^\top,\qquad z=(4,-5,1)^\top.
\]
Then \(x^\top y=0\), \(y^\top z=0\), but \(x^\top z=-9\ne0\).
More generally, if \(x\ne0\) and \(x\perp y\), taking \(z=x\) gives
the first two orthogonality conditions but \(x^\top z=\|x\|^2>0\).
\end{solution}

% MA 3.38*(a), finite real specialization; geometric part (b) omitted.
\begin{problem}[The parallelogram identity \(^{*}\)]\label{ex:06:parallelogram}
An inner product on \(\R^m\) is a function
\(h:\R^m\times\R^m\to\R\) that is symmetric, linear in each variable,
and satisfies \(h(x,x)>0\) for \(x\ne0\).
Linearity gives \(h(0,0)=h(0+0,0)=2h(0,0)\), hence \(h(0,0)=0\).
Write \(\|x\|_h=\sqrt{h(x,x)}\).
Prove, using these properties, that
\[
\|x+y\|_h^2+\|x-y\|_h^2=2\|x\|_h^2+2\|y\|_h^2.
\]
\end{problem}
\begin{solution}
Bilinearity and symmetry give
\[
h(x+y,x+y)=h(x,x)+2h(x,y)+h(y,y),
\]
\[
h(x-y,x-y)=h(x,x)-2h(x,y)+h(y,y).
\]
Adding cancels the two cross terms and proves the identity.
The calculation applies to every such \(h\), including \(h(x,y)=x^\top y\).
\end{solution}

% MA 3.40(a), finite real specialization; complex part (b) omitted.
\begin{problem}[Recovering an inner product]\label{ex:06:polarization}
For an inner product \(h\) as in Exercise~\ref{ex:06:parallelogram},
prove
\[
h(x,y)=\frac14\bigl(\|x+y\|_h^2-\|x-y\|_h^2\bigr).
\]
\end{problem}
\begin{solution}
Subtracting the two expansions in Exercise~\ref{ex:06:parallelogram}
gives
\[
\|x+y\|_h^2-\|x-y\|_h^2=4h(x,y).
\]
Thus two inner products inducing the same norm have the same values
on every pair \(x,y\).
\end{solution}

% MA 3.41*(a,b), complete after correcting m>=2; U6-S02.
\begin{problem}[A norm not induced by an inner product \(^{*}\)]\label{ex:06:l1-norm}
Let \(m\geqslant2\) and put \(\nu(x)=\sum_{i=1}^m|x_i|\).
\begin{enumerate}
\item Prove the three norm properties:
\[
\nu(cx)=|c|\nu(x),\quad
\nu(x)\geqslant0\text{ with equality iff }x=0,\quad
\nu(x+y)\leqslant\nu(x)+\nu(y).
\]
\item Prove that no inner product \(h\) on \(\R^m\) satisfies
\(\nu(x)=\sqrt{h(x,x)}\) for every \(x\).
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item Each term \(|x_i|\) is nonnegative; their sum is zero exactly
when every \(x_i=0\). Also \(|cx_i|=|c||x_i|\), which proves homogeneity.
For real \(s,t\), the inequalities \(-|s|\leqslant s\leqslant|s|\)
and \(-|t|\leqslant t\leqslant|t|\) give
\[
-(|s|+|t|)\leqslant s+t\leqslant|s|+|t|,
\quad\text{hence }|s+t|\leqslant|s|+|t|.
\]
Apply this to each coordinate and add.
\item Take
\[
x=(1,1,0,\ldots,0)^\top,\qquad y=(1,-1,0,\ldots,0)^\top.
\]
Then \(\nu(x)=\nu(y)=\nu(x+y)=\nu(x-y)=2\), so
\[
\nu(x+y)^2+\nu(x-y)^2=8\ne16=2\nu(x)^2+2\nu(y)^2.
\]
This contradicts the necessary identity proved in
Exercise~\ref{ex:06:parallelogram}.
The restriction \(m\geqslant2\) matters: in \(\R\), \(\nu(x)=|x|\)
is induced by \(h(x,y)=xy\).
\end{enumerate}
\end{solution}

% MA 2.39*(a-c), complete; original star verified in scan.
\begin{problem}[The Frobenius norm \(^{*}\)]\label{ex:06:frobenius-norm}
For \(A,B\in\R^{m\times n}\), define
\[
\langle A,B\rangle_F=\sum_{i=1}^m\sum_{j=1}^n a_{ij}b_{ij}
 =\tr(A^\top B),\qquad
\|A\|_F=\left(\sum_{i,j}a_{ij}^2\right)^{1/2}.
\]
Prove
(a) \(\|cA\|_F=|c|\|A\|_F\);
(b) \(\|A\|_F\geqslant0\), with equality iff \(A=0\);
(c) \(\|A+B\|_F\leqslant\|A\|_F+\|B\|_F\).
\end{problem}
\begin{solution}
Let \(a,b\in\R^{mn}\) list the entries of \(A,B\) in the same order,
column by column. Directly from the definitions,
\[
\langle A,B\rangle_F=a^\top b,\quad
\|A\|_F=\|a\|,\quad \|B\|_F=\|b\|,\quad \|A+B\|_F=\|a+b\|.
\]
The trace equality follows because
\((A^\top B)_{jj}=\sum_i a_{ij}b_{ij}\).
The list for \(cA\) is \(ca\), and \(a=0\) iff \(A=0\).
The three assertions therefore follow from the proved vector norm
properties. The same correspondence proves symmetry, bilinearity,
and strict positivity of \(\langle\ ,\ \rangle_F\);
these properties will also be used for matrix subspaces below.
\end{solution}

% MA 3.44(a,b), complete finite real specialization.
\begin{problem}[Zero and an orthogonal complement]\label{ex:06:orthogonal-r3}
\begin{enumerate}
\item Prove that the zero column is orthogonal to every column.
\item Find all columns in \(\R^3\) orthogonal to \((2,2,-1)^\top\).
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item \(0^\top y=\sum_i0y_i=0\) for every \(y\).
\item Writing \(y=(y_1,y_2,y_3)^\top\), the necessary and sufficient
equation is \(2y_1+2y_2-y_3=0\). Thus all solutions, with unrestricted
\(s,t\in\R\), are
\[
y=(s,t,2s+2t)^\top=s(1,0,2)^\top+t(0,1,2)^\top.
\]
Both generators have dot product zero with the given column, verifying
the reverse inclusion.
\end{enumerate}
\end{solution}

\subsection*{Orthonormal bases}

% Lang LA V.2.0, complete.
\begin{problem}[Two orthogonality conditions]\label{ex:06:complement-dimension}
What is the dimension of the subspace of \(\R^6\) orthogonal to both
\[
a=(1,1,-2,3,4,5)^\top,\qquad b=(0,0,1,1,0,7)^\top?
\]
\end{problem}
\begin{solution}
In a relation \(sa+tb=0\), coordinate 1 gives \(s=0\), and coordinate 3
then gives \(t=0\). Hence \(\dim\Span(a,b)=2\).
Being orthogonal to \(a,b\) is equivalent, by linearity, to being
orthogonal to every \(sa+tb\).
The required space is \(\Span(a,b)^\perp\), of dimension \(6-2=4\).
\end{solution}

% Lang LA V.2.1(a,b), complete with original manual calculations.
\begin{problem}[Two bases in \(\R^3\)]\label{ex:06:gs-r3}
Find an orthonormal basis for the span of each list:
\begin{enumerate}
\item \((1,1,-1)^\top,\ (1,0,1)^\top\).
\item \((2,1,1)^\top,\ (1,3,-1)^\top\).
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item Put \(a=(1,1,-1)^\top,b=(1,0,1)^\top\).
Then \(a^\top b=1-1=0\), \(\|a\|^2=3\), and \(\|b\|^2=2\).
The two nonzero orthogonal columns are independent, so
\[
\frac{a}{\sqrt3},\qquad\frac{b}{\sqrt2}
\]
is an orthonormal list with the required span.
\item Put \(a=(2,1,1)^\top,b=(1,3,-1)^\top\).
Here \(a^\top a=6\) and \(a^\top b=4\), so
\[
u=b-\frac46a=\frac13(-1,7,-5)^\top,\qquad
a^\top(-1,7,-5)^\top=-2+7-5=0.
\]
The squared norm of \((-1,7,-5)^\top\) is \(1+49+25=75\).
An answer is
\[
\frac1{\sqrt6}(2,1,1)^\top,\qquad
\frac1{\sqrt{75}}(-1,7,-5)^\top.
\]
Both are nonzero and orthogonal. Since \(u=b-\frac23a\) and
\(b=u+\frac23a\), the new and original lists have the same span.
\end{enumerate}
\end{solution}

% Lang LA V.2.2(a,b), complete.
\begin{problem}[Two bases in \(\R^4\)]\label{ex:06:gs-r4}
Find an orthonormal basis for the span of each list:
\begin{enumerate}
\item \((1,2,1,0)^\top,\ (1,2,3,1)^\top\).
\item \((1,1,0,0)^\top,\ (1,-1,1,1)^\top,\ (-1,0,2,1)^\top\).
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item Call the two given columns \(a,b\), in order.
Then \(a^\top a=6\), \(a^\top b=8\), so
\[
u=b-\frac86a=\frac13(-1,-2,5,3)^\top.
\]
The numerator has squared norm \(1+4+25+9=39\), and its dot product
with \(a\) is \(-1-4+5=0\).
The answer is
\[
\frac1{\sqrt6}(1,2,1,0)^\top,\qquad
\frac1{\sqrt{39}}(-1,-2,5,3)^\top.
\]
The relations \(u=b-\frac43a\), \(b=u+\frac43a\) prove span equality.
\item Write the original columns as \(a,b,c\).
Their relevant products are
\[
a^\top b=0,\quad a^\top a=2,\quad b^\top b=4,\quad
c^\top a=-1,\quad c^\top b=2.
\]
Hence
\[
u=c+\frac12a-\frac12b=\frac12(-2,2,3,1)^\top.
\]
For \(d=(-2,2,3,1)^\top\), \(a^\top d=0\),
\(b^\top d=-2-2+3+1=0\), and \(d^\top d=18\).
An answer is
\[
\frac1{\sqrt2}(1,1,0,0)^\top,\qquad
\frac12(1,-1,1,1)^\top,\qquad
\frac1{\sqrt{18}}(-2,2,3,1)^\top.
\]
The formulas for \(u\) and \(c=u-\frac12a+\frac12b\) prove equality
of the old and new spans. Nonzero orthogonal columns are independent,
so both displayed lists are bases.
\end{enumerate}
\end{solution}

% Harville 6.4(a)(1-3),(b), complete for real matrices with Frobenius inner product.
\begin{problem}[Gram--Schmidt with dependent inputs]\label{ex:06:dependent-gs}
Let \(A_1,\ldots,A_k\in\R^{m\times n}\), \(k\geqslant1\), possibly
dependent or zero. Use the inner product
\(\langle A,B\rangle_F=\tr(A^\top B)\).
\begin{enumerate}
\item Prove all three assertions:
\begin{enumerate}[label=(\arabic*)]
\item Scalars \(c_{ij}\), \(i<j\), can be chosen so that
\[
B_1=A_1,\qquad B_j=A_j-\sum_{i<j}c_{ij}B_i
\quad(2\leqslant j\leqslant k)
\]
are pairwise orthogonal.
\item For every nonzero \(B_i\), orthogonality forces the unique value
\[
c_{ij}=\frac{\langle A_j,B_i\rangle_F}{\langle B_i,B_i\rangle_F}
\qquad(i<j).
\]
\item The number of nonzero \(B_i\)'s equals
\(\dim\Span(A_1,\ldots,A_k)\).
\end{enumerate}
\item Describe a procedure yielding an orthonormal basis of
\(\Span(A_1,\ldots,A_k)\).
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item Suppose \(B_1,\ldots,B_{j-1}\) have been chosen and are pairwise
orthogonal; the assertion is empty for \(j=1\).
For \(i<j\), orthogonality of the preceding matrices gives
\[
\langle B_j,B_i\rangle_F
=\langle A_j,B_i\rangle_F-c_{ij}\|B_i\|_F^2.
\]
If \(B_i\ne0\), its squared norm is positive, so this vanishes for
exactly the stated coefficient. If \(B_i=0\), the equation is \(0=0\);
any coefficient works, and we choose \(c_{ij}=0\).
Induction proves (1) and (2), without division by zero.

For (3), induction on \(j\) using the defining expression for \(B_j\)
shows that each \(B_j\in\Span(A_1,\ldots,A_j)\).
Conversely
\[
A_j=B_j+\sum_{i<j}c_{ij}B_i
\]
puts each \(A_j\) in \(\Span(B_1,\ldots,B_j)\).
Thus every prefix has the same span.
If a linear combination of the nonzero \(B_i\)'s is zero, taking its
inner product with \(B_\ell\ne0\) gives its coefficient times
\(\|B_\ell\|_F^2=0\); each coefficient is therefore zero.
The nonzero outputs form a basis, proving the dimension assertion.
\item Compute these residuals, discard exactly the zero outputs, and
divide every remaining \(B_i\) by \(\|B_i\|_F\).
This preserves their span and makes the basis orthonormal.
If every input is zero, every output is zero and the resulting empty
list is the basis of the zero space.
\end{enumerate}
\end{solution}

\subsection*{Orthogonal matrices and ranks}

% MA 4.23(b,c); original (a) is already the Unit 2 square inverse criterion.
\begin{problem}[Rectangular orthonormality]\label{ex:06:rectangular-orthogonality}
\begin{enumerate}
\item Let \(A\in\R^{m\times n}\), with \(m<n\). Prove that no
\(B\in\R^{m\times n}\) can satisfy \(B^\top A=I_n\).
\item Give an \(m\times n\) example, \(m\ne n\), satisfying
\(AA^\top=I_m\) or \(A^\top A=I_n\), but not both.
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item If \(B^\top A=I_n\), the product-rank bound would give
\[
n=\rank(B^\top A)\leqslant\rank A\leqslant m<n.
\]
This is impossible.
\item Take \(n>1\), let \(e_1\) be the first standard basis column of
\(\R^n\), and set \(A=e_1^\top\), so \(m=1\).
Then \(AA^\top=e_1^\top e_1=I_1\). However,
\(A^\top A=e_1e_1^\top\ne I_n\), since its \((2,2)\) entry is zero.
Transposing this example gives the other alternative.
\end{enumerate}
\end{solution}

% MA 4.27(a-d), complete.
\begin{problem}[Orthogonal matrices]\label{ex:06:orthogonal-matrices}
Let \(A\in\R^{n\times n}\) satisfy \(A^\top A=I_n\).
Prove
(a) \(A\) is invertible;
(b) \(A^{-1}=A^\top\);
(c) \(A^\top\) is orthogonal;
(d) \(AB\) is orthogonal whenever \(B\in\R^{n\times n}\) is orthogonal.
\end{problem}
\begin{solution}
For (a) and (b), the square one-sided inverse criterion of Unit 2
applied to \(A^\top A=I_n\) gives \(A^{-1}=A^\top\) and \(AA^\top=I_n\).
For (c), writing \(C=A^\top\) gives \(C^\top C=AA^\top=I_n\).
For (d),
\[
(AB)^\top(AB)=B^\top A^\top AB=B^\top B=I_n.
\]
For completeness, the reverse product is also
\((AB)(AB)^\top=ABB^\top A^\top=AA^\top=I_n\).
\end{solution}

% MA 8.4(a-c), complete; quantify part (b) and include n>1 in the original counterexample.
\begin{problem}[Preserving one norm or every norm]\label{ex:06:norm-preservation}
\begin{enumerate}
\item Prove that an orthogonal \(Q\in\R^{n\times n}\) satisfies
\(\|Qx\|=\|x\|\) for every \(x\in\R^n\).
\item For a particular nonzero \(x\), does \(\|Qx\|=\|x\|\) imply
that \(Q\) is orthogonal?
\item Suppose a fixed square \(Q\), independent of \(x\), satisfies
\(\|Qx\|=\|x\|\) for every \(x\in\R^n\). Prove that \(Q\) is orthogonal.
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item \(\|Qx\|^2=x^\top Q^\top Qx=x^\top x=\|x\|^2\);
both norms are nonnegative, so they are equal.
\item For \(n>1,x\ne0\), take
\[
Q=\frac{xx^\top}{x^\top x}.
\]
Then \(Qx=x\), \(Q^\top=Q\), and \(Q^2=Q\), by direct multiplication.
Its columns are multiples of \(x\), and \(Qx=x\ne0\); hence \(\rank Q=1<n\).
Thus \(Q^\top Q=Q\ne I_n\), so \(Q\) is not orthogonal.
For \(n=1\), however, the answer is yes: if \(x\ne0\) and
\(|qx|=|x|\), then \(|q|=1\), hence \(q^2=1\).
\item Put \(H=I_n-Q^\top Q\), a fixed symmetric matrix.
The hypothesis gives \(x^\top Hx=0\) for every \(x\).
Taking \(x=e_i\) gives \(h_{ii}=0\).
For \(i\ne j\), taking \(x=e_i+e_j\) then gives
\[
0=h_{ii}+h_{ij}+h_{ji}+h_{jj}=2h_{ij}.
\]
Every entry of \(H\) is zero, so \(Q^\top Q=I_n\).
\end{enumerate}
\end{solution}

% MA 4.18*, complete (not MA 4.25, whose full-rank special cases occur in Unit 4).
\begin{problem}[Rank equalities \(^{*}\)]\label{ex:06:rank-equalities}
For \(A\in\R^{m\times n}\), \(B\in\R^{n\times p}\), prove
\[
\rank(A^\top AB)=\rank(AB)=\rank(ABB^\top).
\]
\end{problem}
\begin{solution}
Proposition~\ref{prop:06:gram} gives \(\rank(X^\top X)=\rank X\).
Applying it to \(X^\top\), and using the equality of row and column
ranks, also gives
\(\rank(XX^\top)=\rank(X^\top)=\rank X\).
The product-rank bounds now yield
\[
\rank(AB)
=\rank(B^\top A^\top AB)
\leqslant\rank(A^\top AB)
\leqslant\rank(AB),
\]
and
\[
\rank(AB)
=\rank(ABB^\top A^\top)
\leqslant\rank(ABB^\top)
\leqslant\rank(AB).
\]
Each middle rank is trapped between two equal quantities.
No full-rank assumption on \(A\) or \(B\) is required.
\end{solution}

\subsection*{Identifying projection matrices}

% MA 8.64(a,b), complete; replace the source trace-rank proof in (b) by a direct sum.
\begin{problem}[A sum of idempotents]\label{ex:06:idempotent-sum}
Let \(A,B\in\R^{n\times n}\) satisfy \(A^2=A,B^2=B\).
\begin{enumerate}
\item Prove that \(A+B\) is idempotent if and only if \(AB=BA=0\).
\item If \(A+B\) is idempotent, prove
\(\rank(A+B)=\rank A+\rank B\).
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item If \(AB=BA=0\), expanding gives \((A+B)^2=A+B\).
Conversely, idempotence of the sum gives \(AB+BA=0\).
Multiplying this equation on the left and on the right by \(A\) gives
\[
AB+ABA=0,\qquad ABA+BA=0.
\]
Subtracting gives \(AB=BA\). Thus \(2AB=0\), and over \(\R\)
we obtain \(AB=BA=0\).
\item Put \(U=\mathcal C(A)\), \(V=\mathcal C(B)\).
Certainly \(\mathcal C(A+B)\subseteq U+V\).
For every \(u=Ax\), \(v=By\), part (a) gives
\[
(A+B)(u+v)=A^2x+BAx+ABy+B^2y=u+v.
\]
Hence \(U+V\subseteq\mathcal C(A+B)\).
If \(z=Ax=By\in U\cap V\), then \(Az=z\), but also \(Az=ABy=0\);
therefore \(z=0\).
The sum is direct, so the dimension formula gives
\[
\rank(A+B)=\dim(U\oplus V)=\dim U+\dim V.
\]
\end{enumerate}
\end{solution}

% MA 8.67, complete; first original proof, with invertibility and empty-block details.
\begin{problem}[Two complementary column blocks]\label{ex:06:complementary-blocks}
Let \(A=[\,A_1\ A_2\,]\) be \(n\times n\).
Suppose
\[
\rank A_1+\rank A_2=n,\qquad A_1^\top A_2=0.
\]
Prove that the two Gram inverses in the following expression exist,
and that
\[
A_1(A_1^\top A_1)^{-1}A_1^\top+
A_2(A_2^\top A_2)^{-1}A_2^\top=I_n.
\]
If a block has no columns, interpret its entire term as the zero matrix.
\end{problem}
\begin{solution}
If \(n=0\), both terms are the empty zero matrix by the stated convention,
so the identity holds. Now let \(n>0\).
If one block is empty, the other is a square full-rank matrix \(C\),
hence invertible. The inverse product rule gives
\[
C(C^\top C)^{-1}C^\top
=CC^{-1}(C^\top)^{-1}C^\top=I_n.
\]
Together with the zero term from the empty block, this proves the formula.

Now suppose both blocks are nonempty, with \(n_i\) columns in \(A_i\).
Since \(n_1+n_2=n\) and
\(\rank A_i\leqslant n_i\), the rank-sum hypothesis forces
\(\rank A_i=n_i\) for both \(i\).
Thus \(G_i=A_i^\top A_i\) is invertible by
Proposition~\ref{prop:06:gram}.
If \(A_1x+A_2y=0\), multiplication by \(A_1^\top\) gives \(G_1x=0\),
so \(x=0\); full column rank of \(A_2\) then gives \(y=0\).
Hence \(A\) is invertible.

Orthogonality and transposition give
\[
A^\top A=\begin{pmatrix}G_1&0\\0&G_2\end{pmatrix},\qquad
(A^\top A)^{-1}=\begin{pmatrix}G_1^{-1}&0\\0&G_2^{-1}\end{pmatrix};
\]
the inverse formula is checked by multiplication in both orders.
The inverse product rule gives \((A^\top A)^{-1}=A^{-1}(A^\top)^{-1}\).
Therefore
\[
I_n=A(A^\top A)^{-1}A^\top
=A_1G_1^{-1}A_1^\top+A_2G_2^{-1}A_2^\top.
\]
\end{solution}

\begin{problem}[Two projection criteria]\label{ex:06:projection-criteria}
Let \(A\in\R^{n\times n}\) be symmetric and idempotent, and let
\(B\in\R^{n\times m}\) have rank \(m\).
\begin{enumerate}
\item If \(AB=B\) and \(\rank A=\rank B\), prove
\[
A=B(B^\top B)^{-1}B^\top.
\]
\item If \(AB=0\) and \(\rank A+\rank B=n\), prove
\[
A=I_n-B(B^\top B)^{-1}B^\top.
\]
\end{enumerate}
\end{problem}
\begin{solution}
The Gram inverse exists by Proposition~\ref{prop:06:gram}.
\begin{enumerate}
\item The equation \(AB=B\) shows that \(\mathcal C(B)\subseteq\mathcal C(A)\).
Their dimensions are equal, so these spaces coincide.
Proposition~\ref{prop:06:projector} says \(A\) projects orthogonally
onto \(\mathcal C(A)\); Proposition~\ref{prop:06:full-rank} identifies
\(B(B^\top B)^{-1}B^\top\) as the projection onto \(\mathcal C(B)\).
Uniqueness of the projection of every vector gives equality of the matrices.
\item Put \(C=I_n-A\). Directly, \(C^\top=C\) and \(C^2=C\), while
\(CB=B\).
By Proposition~\ref{prop:06:projector},
\(\mathcal C(C)=\mathcal C(A)^\perp\); the complement dimension formula gives
\[
\rank C=n-\rank A=\rank B.
\]
Part (a), applied to \(C,B\), gives
\(C=B(B^\top B)^{-1}B^\top\), which is the required formula.
\end{enumerate}
\end{solution}

\subsection*{Inequalities and equality conditions}

% MA 12.1(a,b); original (c), using PSD matrices, is not selected.
\begin{problem}[Two matrix proofs of Schwarz]\label{ex:06:matrix-schwarz}
For \(a,b\in\R^n\), prove
\[
(a^\top b)^2\leqslant(a^\top a)(b^\top b),
\]
with equality exactly when \(a,b\) are linearly dependent.
Give a proof using each of the following matrices:
\begin{enumerate}
\item \(H=ab^\top-ba^\top\).
\item \(P=I_n-bb^\top/(b^\top b)\), for \(b\ne0\), treating \(b=0\) separately.
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item Expanding each entry gives
\[
\begin{aligned}
\|H\|_F^2
&=\sum_{i=1}^n\sum_{j=1}^n(a_i b_j-b_i a_j)^2\\
&=\left(\sum_i a_i^2\right)\left(\sum_j b_j^2\right)
 +\left(\sum_i b_i^2\right)\left(\sum_j a_j^2\right)
 -2\left(\sum_i a_i b_i\right)\left(\sum_j a_j b_j\right)\\
&=2(a^\top a)(b^\top b)-2(a^\top b)^2\geqslant0.
\end{aligned}
\]
Equality holds exactly when \(a_i b_j=b_i a_j\) for every \(i,j\).
If \(b\ne0\), choose \(j\) with \(b_j\ne0\); these equations give
\(a_i=(a_j/b_j)b_i\) for every \(i\), so \(a\) is a multiple of \(b\).
If \(b=0\), equality holds for every \(a\).
Conversely, either \(b=0\) or \(a=cb\) gives \(H=0\).
These are exactly the dependent pairs.
\item If \(b=0\), both sides are zero.
For \(b\ne0\), put \(E=bb^\top/(b^\top b)\).
Then \(E^\top=E\) and
\[
E^2=\frac{b(b^\top b)b^\top}{(b^\top b)^2}=E.
\]
Consequently \(P^\top=P\), \(P^2=P\), and
\[
(a^\top a)(b^\top b)-(a^\top b)^2
=(b^\top b)a^\top Pa
=(b^\top b)\|Pa\|^2\geqslant0.
\]
Equality is equivalent to \(Pa=0\), or
\(a=((b^\top a)/(b^\top b))b\).
Conversely, this relation makes \(Pa=0\).
\end{enumerate}
\end{solution}

\begin{problem}[Three trace inequalities]\label{ex:06:trace-inequalities}
For \(A,B\in\R^{m\times n}\), prove all three inequalities and their
equality conditions:
\begin{enumerate}
\item
\[
\bigl(\tr(A^\top B)\bigr)^2
\leqslant\tr(A^\top A)\tr(B^\top B),
\]
with equality iff one of \(A,B\) is a scalar multiple of the other.
\item
\[
\tr\bigl((A^\top B)^2\bigr)\leqslant\tr(A^\top AB^\top B),
\]
with equality iff \(AB^\top\) is symmetric.
\item
\[
\tr\bigl((A^\top B)^2\bigr)\leqslant\tr(AA^\top BB^\top),
\]
with equality iff \(A^\top B\) is symmetric.
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}
\item List the entries of \(A,B\) in the same order as \(a,b\in\R^{mn}\).
Then the three traces are \(a^\top b,a^\top a,b^\top b\).
Theorem~\ref{thm:06:schwarz} gives the inequality, with equality iff
\(b=0\) or \(a=cb\) for some \(c\in\R\).
In matrix terms this is \(B=0\) or \(A=cB\), exactly the stated condition
that one is a multiple of the other.
\item For any real square \(C\), the trace laws give
\[
\begin{aligned}
\|C-C^\top\|_F^2
&=\tr(C^\top C)-\tr((C^\top)^2)-\tr(C^2)+\tr(CC^\top)\\
&=2\bigl[\tr(C^\top C)-\tr(C^2)\bigr].
\end{aligned}
\]
Thus \(\tr(C^2)\leqslant\tr(C^\top C)\), with equality iff \(C=C^\top\).
Take \(C=AB^\top\). By cyclicity and transposition,
\[
\tr(C^2)=\tr(B^\top AB^\top A)
=\tr\bigl((A^\top B)^2\bigr),
\]
\[
\tr(C^\top C)=\tr(BA^\top AB^\top)=\tr(A^\top AB^\top B).
\]
The preceding identity proves (b), including its exact equality condition.
\item Apply the same identity to \(C=A^\top B\). Now
\[
\tr(C^\top C)=\tr(B^\top AA^\top B)=\tr(AA^\top BB^\top).
\]
Equality holds exactly when this \(C\) is symmetric, proving (c).
\end{enumerate}
\end{solution}

\begin{problem}[A bound for a sum]\label{ex:06:sum-bound}
For \(x_1,\ldots,x_n\in\R\), \(n\geqslant1\), prove
\[
\left(\sum_{i=1}^n x_i\right)^2\leqslant n\sum_{i=1}^n x_i^2,
\]
with equality iff \(x_1=\cdots=x_n\).
\end{problem}
\begin{solution}
Let \(\boldsymbol1=(1,\ldots,1)^\top\in\R^n\) and
\(M=I_n-\boldsymbol1\boldsymbol1^\top/n\).
Since \(\boldsymbol1^\top\boldsymbol1=n\), multiplication gives
\[
M^\top=M,\qquad
M^2=I_n-\frac2n\boldsymbol1\boldsymbol1^\top
+\frac1n\boldsymbol1\boldsymbol1^\top=M.
\]
Therefore
\[
0\leqslant\|Mx\|^2=x^\top Mx
=\sum_i x_i^2-\frac1n\left(\sum_i x_i\right)^2.
\]
Multiplying by \(n>0\) proves the bound.
Equality holds iff \(Mx=0\), or
\(x_i=(\sum_jx_j)/n\) for every \(i\), which is equivalent to all
the entries being equal.
\end{solution}

\subsection*{Matrix subspaces and least squares}

% Harville 12.2, complete; Frobenius inner product specified.
\begin{problem}[A dimension obstruction]\label{ex:06:matrix-orthogonality}
Let \(U,V\) be subspaces of \(\R^{m\times n}\), with inner product
\(\langle A,B\rangle_F=\tr(A^\top B)\).
Prove that if \(\dim V>\dim U\), then \(V\) contains a nonzero matrix
orthogonal to every member of \(U\).
\end{problem}
\begin{solution}
Take bases \(A_1,\ldots,A_r\) of \(U\) and \(B_1,\ldots,B_s\) of \(V\),
where \(s>r\).
If \(r=0\), any \(B_1\ne0\) has the required property, since \(U=\{0\}\).
If \(r>0\), define \(H\in\R^{r\times s}\) by
\[
h_{ij}=\langle A_i,B_j\rangle_F.
\]
There are more unknowns than equations in \(Hx=0\), so a nonzero
solution \(x=(x_1,\ldots,x_s)^\top\) exists.
Put \(C=\sum_{j=1}^s x_jB_j\in V\).
Independence of the \(B_j\)'s gives \(C\ne0\), and
\[
\langle A_i,C\rangle_F=\sum_j h_{ij}x_j=(Hx)_i=0
\qquad(1\leqslant i\leqslant r).
\]
Every \(A\in U\) is a linear combination of the \(A_i\)'s;
bilinearity then gives \(\langle A,C\rangle_F=0\), as required.
\end{solution}

\begin{problem}[Projecting a matrix column by column]\label{ex:06:matrix-projection}
Let \(U\subseteq\R^m\) be a subspace and define
\[
\mathcal M=\{[\,w_1\ \cdots\ w_n\,]:w_j\in U\text{ for every }j\}
\subseteq\R^{m\times n}.
\]
Let \(Y\in\R^{m\times n}\), and let \(X\in\R^{m\times p}\)
have columns spanning \(U\), without assuming they are independent.
Use \(\langle A,B\rangle_F=\tr(A^\top B)\).
Here the orthogonal projection is the matrix \(Z\in\mathcal M\) with
\(\langle Y-Z,W\rangle_F=0\) for every \(W\in\mathcal M\).
Show that this projection exists, is unique, and satisfies
\[
Z=XB_*,\qquad\text{for every solution }B_*\in\R^{p\times n}
\text{ of }X^\top XB=X^\top Y.
\]
\end{problem}
\begin{solution}
The zero matrix is in \(\mathcal M\), and linear combinations of its
members stay in \(\mathcal M\), because this holds for each column in \(U\).
Thus \(\mathcal M\) is a subspace.
Write \(Y=[\,y_1\ \cdots\ y_n\,]\), put \(v_j=P_Uy_j\), and
let \(V=[\,v_1\ \cdots\ v_n\,]\in\mathcal M\).
For every \(W=[\,w_1\ \cdots\ w_n\,]\in\mathcal M\),
\[
\langle Y-V,W\rangle_F
=\sum_{j=1}^n(y_j-v_j)^\top w_j=0.
\]
Consequently, for any \(W\in\mathcal M\), the two terms in
\(Y-W=(Y-V)+(V-W)\) are orthogonal, and expansion gives
\[
\|Y-W\|_F^2=\|Y-V\|_F^2+\|V-W\|_F^2.
\]
It follows that \(V\) is the unique closest matrix.
If another \(Z'\in\mathcal M\) has residual orthogonal to \(\mathcal M\),
put \(D=Z'-V\in\mathcal M\). Then
\[
\|D\|_F^2
=\langle Y-V,D\rangle_F-\langle Y-Z',D\rangle_F=0,
\]
so \(D=0\) and \(Z'=V\).
This also proves uniqueness under the defining projection conditions;
thus \(Z=V\).

By Theorem~\ref{thm:06:normal}, each system
\(X^\top Xb_j=X^\top y_j\) is consistent, even if \(X\) has dependent
columns. Stacking one solution of each system gives a solution \(B\).
For every solution \(B_*\), the same theorem gives
\(Xb_{*j}=P_Uy_j=v_j\) for all \(j\).
Therefore \(XB_*=V=Z\), independently of which coefficient solution is used.
\end{solution}

\begin{problem}[Recognizing an orthonormal basis]\label{ex:06:parseval-converse}
Let \(V\subseteq\R^n\) be a subspace and let \(v_1,\ldots,v_r\in V\)
be an orthonormal list. Suppose that
\[
\|v\|^2=\sum_{i=1}^r(v^\top v_i)^2
\qquad\text{for every }v\in V.
\]
Prove that \(v_1,\ldots,v_r\) is a basis of \(V\).
\end{problem}
\begin{solution}
Orthonormal columns are independent by Proposition~\ref{prop:06:orthogonal-list}.
For \(v\in V\), put \(w=\sum_i(v^\top v_i)v_i\).
Orthogonality gives
\[
w^\top w=\sum_i(v^\top v_i)^2,\qquad
v^\top w=\sum_i(v^\top v_i)^2.
\]
The assumed equality now yields
\[
\|v-w\|^2=\|v\|^2-2v^\top w+w^\top w=0.
\]
Thus \(v=w\) is in the span, proving that the list is a basis.
If \(r=0\), the assumption says \(\|v\|^2=0\) for every \(v\in V\);
then \(V=\{0\}\) and the empty list is its basis.
\end{solution}
