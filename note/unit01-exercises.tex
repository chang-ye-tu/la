\subsection*{Matrix multiplication and transpose}

\begin{problem}\label{ex:01:rectangular-associativity}
Let
\[
A=\begin{pmatrix}2&1&-1\\3&1&2\end{pmatrix},\qquad
B=\begin{pmatrix}1&1\\2&0\\3&-1\end{pmatrix},\qquad
c=\begin{pmatrix}1\\3\end{pmatrix}.
\]
Compute $(AB)c$ and $A(Bc)$, showing the intermediate products.
\end{problem}
\begin{solution}
The intermediate products have sizes $2\times2$ and $3\times1$:
\[
AB=\begin{pmatrix}2+2-3&2+1\\3+2+6&3-2\end{pmatrix}
=\begin{pmatrix}1&3\\11&1\end{pmatrix},
\qquad Bc=\begin{pmatrix}4\\2\\0\end{pmatrix}.
\]
Consequently,
\[
(AB)c=\begin{pmatrix}1+9\\11+3\end{pmatrix}
=\begin{pmatrix}10\\14\end{pmatrix},
\qquad
A(Bc)=\begin{pmatrix}8+2\\12+2\end{pmatrix}
=\begin{pmatrix}10\\14\end{pmatrix}.
\]
\end{solution}

\begin{problem}\label{ex:01:reshuffle-order}\notetarget{note:01:exercise-shuffles}
On an ordered triple $(a_1,a_2,a_3)$, let $S$ reverse the order and let
$T$ move the last entry to the first position. Show that performing $S$
then $T$ need not give the same result as performing $T$ then $S$.
Do this first with triples, then with matrices acting on column vectors.
\end{problem}
\begin{solution}\notetarget{note:01:exercise-shuffles-solution}
Applying $S$ then $T$ gives
$(a_1,a_2,a_3)\mapsto(a_3,a_2,a_1)\mapsto(a_1,a_3,a_2)$.
Applying $T$ then $S$ gives
$(a_1,a_2,a_3)\mapsto(a_3,a_1,a_2)\mapsto(a_2,a_1,a_3)$.
For instance, the triple $(1,2,3)$ gives different results.
\notetarget{note:01:exercise-shuffles-matrices}The corresponding matrices are
\[
S=\begin{pmatrix}0&0&1\\0&1&0\\1&0&0\end{pmatrix},\qquad
T=\begin{pmatrix}0&0&1\\1&0&0\\0&1&0\end{pmatrix}.
\]
\notetarget{note:01:exercise-shuffles-products}The two orders give
\[
TS=\begin{pmatrix}1&0&0\\0&0&1\\0&1&0\end{pmatrix},\qquad
ST=\begin{pmatrix}0&1&0\\1&0&0\\0&0&1\end{pmatrix},
\]
which are unequal.
\end{solution}

\begin{problem}\label{ex:01:matrix-constructions}
Construct real $2\times2$ matrices satisfying each condition:
\begin{enumerate}
\item $A^2=-I$;
\item $B^2=0$ but $B\ne0$;
\item $CD=-DC$ and $CD\ne0$;
\item $EF=0$, with every entry of both $E$ and $F$ nonzero.
\end{enumerate}
\end{problem}
\begin{solution}
(a) Take $A=\begin{pmatrix}0&-1\\1&0\end{pmatrix}$. Then
$A^2=\begin{pmatrix}-1&0\\0&-1\end{pmatrix}=-I$.

(b) Take $B=\begin{pmatrix}0&1\\0&0\end{pmatrix}$. It is nonzero,
and direct multiplication gives $B^2=\begin{pmatrix}0&0\\0&0\end{pmatrix}$.

(c) Take
\[
C=\begin{pmatrix}1&0\\0&-1\end{pmatrix},\qquad
D=\begin{pmatrix}0&1\\1&0\end{pmatrix}.
\]
Then $CD=\begin{pmatrix}0&1\\-1&0\end{pmatrix}$ and
$DC=\begin{pmatrix}0&-1\\1&0\end{pmatrix}=-CD$.

(d) Take $E=\begin{pmatrix}1&1\\1&1\end{pmatrix}$ and
$F=\begin{pmatrix}1&-1\\-1&1\end{pmatrix}$. Every entry is nonzero,
but each entry of $EF$ is either $1-1$ or $-1+1$, so $EF=0$.
\end{solution}

\begin{problem}\label{ex:01:difference-of-squares}
Two square matrices $A,B$ of the same size \emph{commute} if $AB=BA$.
Prove that
\[
(A+B)(A-B)=A^2-B^2
\]
holds exactly when $A$ and $B$ commute.
\end{problem}
\begin{solution}
Distributivity gives
\[
(A+B)(A-B)=A^2-AB+BA-B^2.
\]
This equals $A^2-B^2$ exactly when $-AB+BA=0$, or $AB=BA$.
\end{solution}

\begin{problem}\label{ex:01:symmetric-product}
\begin{enumerate}
\item For symmetric $n\times n$ matrices $A,B$, prove that $AB$ is
symmetric exactly when $AB=BA$.
\item Give two symmetric matrices whose product is not symmetric.
\end{enumerate}
\end{problem}
\begin{solution}
(a) Since $A^\top=A$ and $B^\top=B$,
\[
(AB)^\top=B^\top A^\top=BA.
\]
Thus $(AB)^\top=AB$ exactly when $BA=AB$.

(b) The symmetric matrices
$A=\begin{pmatrix}1&0\\0&2\end{pmatrix}$ and
$B=\begin{pmatrix}0&1\\1&0\end{pmatrix}$ give
$AB=\begin{pmatrix}0&1\\2&0\end{pmatrix}$, which is not symmetric.
\end{solution}

\begin{problem}\label{ex:01:symmetric-decomposition}
A real square matrix $K$ is \emph{skew-symmetric} if
$K^\top=-K$. Let $A$ be a real square matrix.
\begin{enumerate}
\item Prove that $A-A^\top$ is skew-symmetric.
\item Express $A$ as the sum of a symmetric matrix and a skew-symmetric
matrix.
\item Prove that every diagonal entry of a skew-symmetric matrix is zero.
\end{enumerate}
\end{problem}
\begin{solution}
(a) Entrywise transposition gives
$(A-A^\top)^\top=A^\top-A=-(A-A^\top)$.

(b) Set
\[
H=\frac{A+A^\top}2,\qquad K=\frac{A-A^\top}2.
\]
Then $A=H+K$, $H^\top=H$, and $K^\top=-K$.

(c) If $K^\top=-K$, the $(i,i)$ entry gives $k_{ii}=-k_{ii}$.
Hence $2k_{ii}=0$, so $k_{ii}=0$.
\end{solution}

\begin{problem}\label{ex:01:outer-product-symmetry}
For real columns $a,b$ of the same length, determine exactly when
$ab^\top=ba^\top$.
\end{problem}
\begin{solution}
If $b=0$, both matrices are zero. Suppose $b\ne0$ and choose an index
$j$ with $b_j\ne0$. Equality of the $(i,j)$ entries gives
$a_i b_j=b_i a_j$ for every $i$, so
\[
a=\frac{a_j}{b_j}b.
\]
Conversely, if $a=kb$ for a real $k$, then
$ab^\top=kbb^\top=ba^\top$.
The exact condition is therefore $b=0$ or $a=kb$ for some real $k$;
this includes $a=0$.
\end{solution}

\begin{problem}\label{ex:01:outer-product-sums}
Let $A=[\,a_1\ \cdots\ a_n\,]$ be an $m\times n$ real matrix,
with entries $a_{rj}$. Prove
\[
AA^\top=\sum_{j=1}^n a_j a_j^\top,
\qquad
(A^\top A)_{ij}=\sum_{r=1}^m a_{ri}a_{rj}.
\]
\end{problem}
\begin{solution}
The $(r,s)$ entry of the first product is
\[
(AA^\top)_{rs}=\sum_{j=1}^n a_{rj}a_{sj}
=\sum_{j=1}^n(a_j a_j^\top)_{rs}.
\]
For the second product, $(A^\top)_{ir}=a_{ri}$, so
\[
(A^\top A)_{ij}
=\sum_{r=1}^m(A^\top)_{ir}A_{rj}
=\sum_{r=1}^m a_{ri}a_{rj}.
\]
Both identities follow from the finite-sum definition of multiplication.
\end{solution}

\begin{problem}[$\ast$]\label{ex:01:gram-product-cancellation}
Let $A\in\R^{m\times n}$ and $B,C\in\R^{n\times p}$. Prove:
\begin{enumerate}
\item $A^\top A=0$ if and only if $A=0$;
\item $AB=0$ if and only if $A^\top AB=0$;
\item $AB=AC$ if and only if $A^\top AB=A^\top AC$.
\end{enumerate}
\end{problem}
\begin{solution}
(a) If $A^\top A=0$, then, for every column $j$,
\[
0=(A^\top A)_{jj}=\sum_{i=1}^m a_{ij}^2.
\]
A sum of squares of real numbers is zero only when every summand is
zero. Thus all entries of $A$ vanish. The converse follows by multiplication.

(b) The forward implication follows by multiplying $AB=0$ on the left
by $A^\top$. Conversely, if $A^\top AB=0$, then
\[
(AB)^\top(AB)=B^\top A^\top AB=0.
\]
Part (a), applied to $AB$, gives $AB=0$.

(c) Apply part (b) with $B-C$ in place of $B$:
\[
AB=AC\ \Longleftrightarrow\ A(B-C)=0
\ \Longleftrightarrow\ A^\top A(B-C)=0
\ \Longleftrightarrow\ A^\top AB=A^\top AC.
\]
\end{solution}

\begin{problem}\label{ex:01:centering-matrix}
Let $n\geqslant1$, let $j=(1,\ldots,1)^\top\in\R^n$, and set
\[
P=I_n-\frac1n jj^\top.
\]
Prove that $P$ is symmetric and \emph{idempotent}, meaning $P^2=P$.
Explain these properties by describing $Px$ for a column $x\in\R^n$.
\end{problem}
\begin{solution}
Since $j^\top j=n$, associativity gives
\[
(jj^\top)^2=j(j^\top j)j^\top=n jj^\top.
\]
Therefore
\[
P^2=I_n-\frac2n jj^\top+\frac1{n^2}(jj^\top)^2
=I_n-\frac1n jj^\top=P.
\]
Also $(jj^\top)^\top=jj^\top$, so
$P^\top=P$.
If $\bar x=\frac1n\sum_{i=1}^n x_i$, then
\[
Px=x-\bar xj.
\]
Thus $P$ subtracts the mean from each coordinate. The result has
coordinate sum $\sum_i(x_i-\bar x)=0$, so applying $P$ again changes
nothing. Its diagonal entries are $1-1/n$ and its off-diagonal entries
are $-1/n$: each coordinate contributes the same amount to every other
coordinate, giving symmetry.
\end{solution}

\subsection*{Solving systems and describing all solutions}

\begin{problem}\label{ex:01:two-equations-three-unknowns}
For each system, write its coefficient matrix, reduce it to row echelon
form, and find all solutions.
\begin{enumerate}[label=(\alph*)]
\item $2x-3y+4z=0$, $3x+y+z=0$.
\item $-x+2y-4z+w=0$, $x+3y+z-w=0$.
\end{enumerate}
\end{problem}
\begin{solution}
(a) The coefficient matrix and a row reduction are
\[
\begin{pmatrix}2&-3&4\\3&1&1\end{pmatrix}
\xrightarrow{R_2\leftarrow R_2-\frac32R_1}
\begin{pmatrix}2&-3&4\\0&11/2&-5\end{pmatrix}.
\]
The second row gives $y=10z/11$. Substituting into the first gives
\[
2x=3y-4z=\frac{30}{11}z-4z=-\frac{14}{11}z,
\qquad x=-\frac7{11}z.
\]
Writing $z=11t$, all solutions are $t(-7,10,11)^\top$, $t\in\R$.

(b) Here
\[
\begin{pmatrix}-1&2&-4&1\\1&3&1&-1\end{pmatrix}
\xrightarrow{R_2\leftarrow R_2+R_1}
\begin{pmatrix}-1&2&-4&1\\0&5&-3&0\end{pmatrix}.
\]
Thus $y=3z/5$, and the first row gives
$x=2y-4z+w=6z/5-4z+w=w-14z/5$.
With $z=5s$ and $w=t$, all solutions are
\[
(x,y,z,w)^\top=(-14s+t,3s,5s,t)^\top,\qquad s,t\in\R.
\]
In both parts the free coordinates can be chosen arbitrarily, and back
substitution determines the remaining coordinates. Hence the displayed
sets include every solution; substituting the formulas verifies them.
\end{solution}

\begin{problem}\label{ex:01:gaussian-unique-solutions}
Solve both systems by Gaussian elimination:
\[
\text{(a)}\quad
\begin{pmatrix}2&1&-3\\5&2&-6\\3&-1&-4\end{pmatrix}x
=\begin{pmatrix}1\\5\\7\end{pmatrix},
\qquad
\text{(b)}\quad
\begin{pmatrix}2&1&-2\\3&2&2\\5&4&3\end{pmatrix}x
=\begin{pmatrix}10\\1\\4\end{pmatrix}.
\]
\end{problem}
\begin{solution}
(a) Divide the first row of the augmented matrix by $2$, then subtract
$5$ times the new first row from row two and $3$ times it from row three:
\[
\left(\begin{array}{rrr|r}
2&1&-3&1\\5&2&-6&5\\3&-1&-4&7
\end{array}\right)
\longrightarrow
\left(\begin{array}{rrr|r}
1&1/2&-3/2&1/2\\0&-1/2&3/2&5/2\\0&-5/2&1/2&11/2
\end{array}\right).
\]
Multiply row two by $-2$, add $5/2$ times this row to row three,
and divide row three by $-7$. This gives
\[
\left(\begin{array}{rrr|r}
1&1/2&-3/2&1/2\\0&1&-3&-5\\0&0&1&1
\end{array}\right).
\]
Back substitution yields $x_3=1$, $x_2=-5+3=-2$, and
$x_1=1/2-x_2/2+3x_3/2=3$.

(b) Divide row one by $2$ and eliminate the entries below its first entry:
\[
\left(\begin{array}{rrr|r}
2&1&-2&10\\3&2&2&1\\5&4&3&4
\end{array}\right)
\longrightarrow
\left(\begin{array}{rrr|r}
1&1/2&-1&5\\0&1/2&5&-14\\0&3/2&8&-21
\end{array}\right).
\]
Multiply row two by $2$, subtract $3/2$ times this row from row three,
and divide row three by $-7$:
\[
\left(\begin{array}{rrr|r}
1&1/2&-1&5\\0&1&10&-28\\0&0&1&-3
\end{array}\right).
\]
Thus $x_3=-3$, $x_2=-28-10(-3)=2$, and
$x_1=5-x_2/2+x_3=1$.

Each echelon system determines all three coordinates, so the solutions
are unique. Substitution into the original systems gives, respectively,
\[
\begin{pmatrix}6-2-3\\15-4-6\\9+2-4\end{pmatrix}
=\begin{pmatrix}1\\5\\7\end{pmatrix},
\qquad
\begin{pmatrix}2+2+6\\3+4-6\\5+8-9\end{pmatrix}
=\begin{pmatrix}10\\1\\4\end{pmatrix}.
\]
\end{solution}

\begin{problem}\label{ex:01:combining-homogeneous-solutions}
Let $A\in\R^{m\times n}$ and suppose $Ax=Ay=0$.
Prove that $x+y$ also solves the homogeneous system and that $cx$
solves it for every real number $c$.
\end{problem}
\begin{solution}
Distributivity and compatibility with scalar multiplication give
\[
A(x+y)=Ax+Ay=0+0=0,
\qquad
A(cx)=c(Ax)=c0=0.
\]
Thus both asserted columns belong to $N(A)$.
\end{solution}

\begin{problem}\label{ex:01:translated-solutions}
Let $A\in\R^{m\times n}$ and suppose $Ax=b$.
If $Ax'=b$ as well, prove that $x'=x+y$ for some $y$ satisfying $Ay=0$.
Conversely, show that $x+y$ solves the original system whenever $Ay=0$.
\end{problem}
\begin{solution}
Set $y=x'-x$. Then $Ay=Ax'-Ax=b-b=0$ and $x'=x+y$.
Conversely, $A(x+y)=Ax+Ay=b+0=b$. Thus the full solution set is
one particular solution plus all solutions of the homogeneous system.
\end{solution}

\begin{problem}\label{ex:01:affine-combinations-of-solutions}
Let $A\in\R^{m\times n}$, $B\in\R^{m\times p}$, and suppose
$AX_i=B$ for $i=1,\ldots,k$, where every $X_i$ is $n\times p$.
If real numbers $c_1,\ldots,c_k$ satisfy $\sum_i c_i=1$, prove that
$\sum_i c_iX_i$ also solves $AX=B$.
\end{problem}
\begin{solution}
All matrices in the sum have the same size. Distributivity gives
\[
A\left(\sum_{i=1}^k c_iX_i\right)
=\sum_{i=1}^k c_iAX_i
=\sum_{i=1}^k c_iB
=\left(\sum_{i=1}^k c_i\right)B=B.
\]
Thus the asserted matrix is a solution, including when $B=0$.
\end{solution}

\subsection*{Consistency and exceptional parameters}

\begin{problem}\label{ex:01:dependent-equations-parameter}
For which real numbers $k$ does the system
\[
x_1+x_2=1,\qquad 2x_1+2x_2=k
\]
have a solution? When it has a solution, is that solution unique?
\end{problem}
\begin{solution}
Subtracting twice the first equation from the second gives $0=k-2$.
Thus a solution exists exactly when $k=2$. In that case the second
equation repeats the first, and all solutions are $(1-t,t)^\top$, $t\in\R$.
There are infinitely many, so the solution is not unique.
\end{solution}

\begin{problem}\label{ex:01:universal-right-hand-sides}
Can a matrix $A$ be chosen so that $Ax=b$ is inconsistent for every
right-hand side $b$ of the appropriate size?
Can a matrix $A$ be chosen so that $Ax=b$ is consistent for every such
$b$? Justify each answer.
\end{problem}
\begin{solution}
The first answer is no: for every matrix $A$, the right-hand side $b=0$
has the solution $x=0$.
The second answer is yes: choose $A=I_m$. Then $x=b$ solves $Ax=b$
for every $b\in\R^m$.
\end{solution}

\begin{problem}\label{ex:01:homogeneous-parameter-systems}
For each real $k$, find all solutions of the following systems:
\begin{enumerate}
\item $x_1+3x_2=0$, \quad $2x_1+kx_2=0$;
\item $x_1+2x_2=0$, \quad $3x_1+6x_2=0$, \quad $2x_1+kx_2=0$.
\end{enumerate}
\end{problem}
\begin{solution}
(a) Subtracting twice the first equation from the second gives
\[
x_1+3x_2=0,\qquad (k-6)x_2=0.
\]
For $k\ne6$, only $(x_1,x_2)=(0,0)$ is possible.
For $k=6$, choose $x_2=t$ freely; all solutions are
$t(-3,1)^\top$, $t\in\R$.

(b) Subtract three times the first equation from the second and twice
the first from the third:
\[
x_1+2x_2=0,\qquad 0=0,\qquad (k-4)x_2=0.
\]
For $k\ne4$, the only solution is zero. For $k=4$, all solutions
are $t(-2,1)^\top$, $t\in\R$.
These reversible operations prove completeness; direct substitution
checks each displayed family.
\end{solution}

\begin{problem}\label{ex:01:inconsistent-three-equations}\notetarget{note:01:exercise-inconsistency}
Does the following system have a solution?
\[
\begin{aligned}
2x_1+3x_2+x_3&=7,\\
x_1+2x_2-2x_3&=8,\\
3x_1+4x_2+4x_3&=20.
\end{aligned}
\]
\end{problem}
\begin{solution}\notetarget{note:01:exercise-inconsistency-solution}
The coefficient rows satisfy
\[
2(2,3,1)-(1,2,-2)=(3,4,4).
\]
Thus twice the first left-hand side minus the second equals the third
left-hand side. The same operation on the right-hand sides gives
$2(7)-8=6$, not $20$. Subtracting twice row one and adding row two to
row three therefore produces $0=14$. The system is inconsistent.
\end{solution}

\begin{problem}\label{ex:01:two-parameter-consistency}
For real parameters $\alpha,\beta$, consider
\[
\begin{aligned}
\alpha x_1+\beta x_2+2x_3&=1,\\
\alpha x_1+(2\beta-1)x_2+3x_3&=1,\\
\alpha x_1+\beta x_2+(\beta+3)x_3&=2\beta-1.
\end{aligned}
\]
When does a solution exist, and when is it unique?
\end{problem}
\begin{solution}
Subtract the first equation from the second and third. The equivalent
system is
\[
\alpha x_1+\beta x_2+2x_3=1,\qquad
(\beta-1)x_2+x_3=0,\qquad
(\beta+1)x_3=2(\beta-1).
\]
\begin{itemize}
\item $\beta=-1$: the last equation is $0=-4$, so no solution exists.
\item $\beta=1$: the equations reduce to $x_3=0$ and
$\alpha x_1+x_2=1$. For every $\alpha$, all solutions are
$(t,1-\alpha t,0)^\top$, $t\in\R$.
\item $\beta\ne\pm1$: solve the last equation for $x_3$, then the
second for $x_2=-x_3/(\beta-1)$:
\[
x_3=\frac{2(\beta-1)}{\beta+1},\qquad
x_2=-\frac2{\beta+1}.
\]
Substituting into the first equation now gives
\[
\alpha x_1=1-\beta x_2-2x_3
=1+\frac{2\beta}{\beta+1}-\frac{4(\beta-1)}{\beta+1}
=\frac{5-\beta}{\beta+1}.
\]
For $\alpha\ne0$, this determines
$x_1=(5-\beta)/(\alpha(\beta+1))$, so the solution is unique.
For $\alpha=0$, it is consistent exactly when $\beta=5$; then the solutions
are $(t,-1/3,4/3)^\top$, $t\in\R$.
\end{itemize}
Thus uniqueness holds exactly when $\alpha\ne0$ and $\beta\ne\pm1$.
The remaining consistent cases are $\beta=1$ with arbitrary $\alpha$,
and $(\alpha,\beta)=(0,5)$; each has infinitely many solutions.
All other parameter pairs give no solution.
\end{solution}

\Needspace{20\baselineskip}
\subsection*{Column sets and independence}

\begin{problem}\label{ex:01:concrete-column-null-sets}
Let
\[
A=\begin{pmatrix}1&2\\3&4\\5&6\end{pmatrix},
\qquad
u=\begin{pmatrix}1\\0\\-1\end{pmatrix},
\quad v=\begin{pmatrix}0\\1\\2\end{pmatrix}.
\]
Prove the following descriptions, including both directions of each
set equality:
\begin{enumerate}
\item $\mathcal C(A)=\{su+tv:s,t\in\R\}$ and
$N(A^\top)=\{r(1,-2,1)^\top:r\in\R\}$;
\item $\mathcal C(A^\top)=\R^2$ and $N(A)=\{0\}$;
\item $\mathcal C(AA^\top)=\mathcal C(A)$.
\end{enumerate}
\end{problem}
\begin{solution}
(a) For $y=(y_1,y_2)^\top$, put
$s=y_1+2y_2$ and $t=3y_1+4y_2$. Then
\[
Ay=(s,t,5y_1+6y_2)^\top=(s,t,-s+2t)^\top=su+tv.
\]
Conversely, any prescribed $s,t$ arise from
\[
y_1=-2s+t,\qquad y_2=\frac{3s-t}{2}.
\]
Substitution gives $y_1+2y_2=s$ and $3y_1+4y_2=t$, proving both
inclusions in the first equality.

The equation $A^\top x=0$ is
\[
x_1+3x_2+5x_3=0,\qquad 2x_1+4x_2+6x_3=0.
\]
Subtracting twice the first equation from the second gives
$-2x_2-4x_3=0$. Thus $x_2=-2x_3$ and $x_1=x_3$.
Taking $x_3=r$ gives exactly the asserted family.

(b) Given $(s,t)^\top\in\R^2$, choose
\[
x=\left(\frac{3t}{2}-2s,\ s-\frac t2,\ 0\right)^\top.
\]
Then $A^\top x=(s,t)^\top$, so every column in $\R^2$ lies in
$\mathcal C(A^\top)$. The reverse inclusion follows from the size of $A^\top$.

If $Ay=0$, its first two equations are
$y_1+2y_2=0$ and $3y_1+4y_2=0$. Subtracting three times the first
from the second gives $-2y_2=0$, and then $y_1=0$.
Conversely, $A0=0$, so $N(A)=\{0\}$.

(c) For every $z\in\R^3$, $AA^\top z=A(A^\top z)$ belongs to $\mathcal C(A)$.
This proves $\mathcal C(AA^\top)\subseteq \mathcal C(A)$.
For the reverse inclusion, compute
\[
AA^\top=\begin{pmatrix}5&11&17\\11&25&39\\17&39&61\end{pmatrix}
=[\,d_1\ d_2\ d_3\,].
\]
Its first two columns satisfy
\[
\frac{25d_1-11d_2}{4}=u,\qquad
\frac{-11d_1+5d_2}{4}=v.
\]
Consequently every $su+tv$ is a linear combination of $d_1,d_2$.
Part (a) now gives $\mathcal C(A)\subseteq \mathcal C(AA^\top)$.
\end{solution}

\begin{problem}\label{ex:01:same-column-set}
Let $a,b,c\in\R^m$ satisfy $a+b+c=0$.
Prove that
\[
\mathcal C([\,a\ b\,])=\mathcal C([\,a\ c\,]).
\]
\end{problem}
\begin{solution}
Since $b=-a-c$, every column in the first set has the form
\[
sa+tb=sa+t(-a-c)=(s-t)a-tc,
\]
and hence belongs to the second set.
Conversely, $c=-a-b$, so
\[
pa+qc=pa+q(-a-b)=(p-q)a-qb
\]
belongs to the first set. The two inclusions prove the equality.
Neither calculation requires any of the three columns to be nonzero
or independent.
\end{solution}

\begin{problem}\label{ex:01:dependence-relations}
Prove the following statements about finite lists of real column vectors:
\begin{enumerate}
\item The list consisting only of the zero column is linearly dependent.
\item Any list containing the zero column is linearly dependent.
\item A list of nonzero columns $x_1,\ldots,x_n$ is linearly dependent
if and only if one of its columns is a linear combination of the others.
\item A list of nonzero columns $x_1,\ldots,x_n$ is linearly dependent
if and only if some $x_j$, with $2\leqslant j\leqslant n$, is a linear
combination of $x_1,\ldots,x_{j-1}$.
\end{enumerate}
An empty sum is understood to be zero.
\end{problem}
\begin{solution}
(a) The relation $1\cdot0=0$ has a nonzero coefficient.

(b) Give coefficient $1$ to a zero column and coefficient $0$ to every
other column. This is a nontrivial relation.

(c) If $x_j=\sum_{i\ne j}a_ix_i$, subtract $x_j$ to obtain
\[
\sum_{i\ne j}a_ix_i-x_j=0.
\]
Its coefficient of $x_j$ is $-1$, so the list is dependent.
Conversely, take a relation $\sum_i c_ix_i=0$ with at least one
$c_j\ne0$. Solving for that column gives
\[
x_j=-\sum_{i\ne j}\frac{c_i}{c_j}x_i.
\]

(d) A relation $x_j=\sum_{i<j}a_ix_i$ gives a nontrivial relation on
the full list by setting later coefficients to zero.
Conversely, start with a nontrivial relation and let $j$ be the
largest index with $c_j\ne0$. This index cannot be $1$, since then
$c_1x_1=0$ would contradict $x_1\ne0$. All coefficients after $j$
vanish, and therefore
\[
x_j=-\sum_{i=1}^{j-1}\frac{c_i}{c_j}x_i.
\]
For a one-column list with $x_1\ne0$, both sides of each equivalence
in (c) and (d) are false, as required.
\end{solution}

\begin{problem}\label{ex:01:triangular-independence}
\begin{enumerate}
\item Prove that the columns of $I_n$ are linearly independent.
\item Show that the columns of
\[
A=\begin{pmatrix}1&3&2&0\\1&-5&-3&1\\5&-1&0&2\end{pmatrix}
\]
are linearly dependent. Are its rows linearly dependent?
For rows, dependence means a nontrivial linear combination equal to
the zero row.
\item A square matrix is \emph{upper triangular} if its entries below
the diagonal are zero, and \emph{lower triangular} if its entries above
the diagonal are zero. Prove that the columns of a triangular matrix
are linearly independent exactly when every diagonal entry is nonzero.
\end{enumerate}
\end{problem}
\begin{solution}
(a) If $c_1e_1+\cdots+c_ne_n=0$, the $i$th coordinate is $c_i=0$.
Thus all coefficients vanish.

(b) Denote the columns by $a_1,\ldots,a_4$ and the rows by
$r_1,r_2,r_3$. Direct calculation gives nontrivial relations:
\[
3a_1-a_2-8a_4
=\begin{pmatrix}3-3\\3+5-8\\15+1-16\end{pmatrix}=0,
\]
\[
3r_1+2r_2-r_3
=(3+2-5,\ 9-10+1,\ 6-6,\ 2-2)=0.
\]
Hence both lists are dependent.

(c) Let the triangular matrix be $T=(t_{ij})$.
If all diagonal entries are nonzero, back substitution for an upper
triangular matrix gives $x_n=0,x_{n-1}=0,\ldots,x_1=0$ in $Tx=0$.
For a lower triangular matrix, solve from the first row downward
instead. Thus only $x=0$ solves the system, proving column independence.

For the converse, construct a nonzero solution whenever a diagonal entry is zero.
\begin{itemize}
\item Upper triangular: let $k$ be the \emph{first} zero diagonal index.
Set $x_k=1$ and $x_j=0$ for $j>k$. Row $k$ vanishes because $t_{kk}=0$;
rows below it involve only zero coordinates. For $i<k$, we have $t_{ii}\ne0$,
so define
\[
x_i=-\frac{1}{t_{ii}}\sum_{j=i+1}^{k}t_{ij}x_j,
\qquad i=k-1,k-2,\ldots,1.
\]
Each step makes row $i$ vanish, and $x_k=1$ ensures $x\ne0$.
\item Lower triangular: let $k$ be the \emph{last} zero diagonal index.
Set $x_j=0$ for $j<k$ and $x_k=1$. Rows above $k$ involve only zero
coordinates; row $k$ vanishes because $t_{kk}=0$.
All later diagonal entries are nonzero, so define
\[
x_i=-\frac{1}{t_{ii}}\sum_{j=k}^{i-1}t_{ij}x_j,
\qquad i=k+1,k+2,\ldots,n.
\]
Again every row vanishes and $x_k=1$.
\end{itemize}
Thus a zero diagonal entry gives a nonzero solution of $Tx=0$, hence
linearly dependent columns.
\end{solution}

\begin{problem}\label{ex:01:independent-pairwise-sums}
Let $A,B,C$ be linearly independent real $m\times n$ matrices.
Determine whether $A+B$, $A+C$, and $B+C$ are linearly independent.
\end{problem}
\begin{solution}
Suppose
\[
\alpha(A+B)+\beta(A+C)+\gamma(B+C)=0.
\]
Collecting coefficients gives
\[
(\alpha+\beta)A+(\alpha+\gamma)B+(\beta+\gamma)C=0.
\]
Independence of $A,B,C$ implies
\[
\alpha+\beta=0,\qquad
\alpha+\gamma=0,\qquad
\beta+\gamma=0.
\]
The first two equations give $\beta=\gamma=-\alpha$; the last then
gives $-2\alpha=0$. Hence $\alpha=\beta=\gamma=0$.
The three pairwise sums are therefore linearly independent.
\end{solution}
